cmake_minimum_required(VERSION 3.20)
project(Astana3DTour LANGUAGES C CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Existing GLFW requires MSVC x64; use VS 2022 -A x64")
endif()
# Existing GLFW uses MSVCRTD. Keep both targets on /MDd.
set(CMAKE_CONFIGURATION_TYPES Debug CACHE STRING "Configurations" FORCE)
set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDebugDLL)
set(DEPENDENCY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "Workspace dependencies")
foreach(file include/GLFW/glfw3.h include/glad/glad.h include/KHR/khrplatform.h src/glad.c lib/glfw3.lib)
 if(NOT EXISTS "${DEPENDENCY_ROOT}/${file}")
  message(FATAL_ERROR "Missing dependency: ${DEPENDENCY_ROOT}/${file}")
 endif()
endforeach()
add_library(glfw_existing STATIC IMPORTED)
set_target_properties(glfw_existing PROPERTIES IMPORTED_LOCATION "${DEPENDENCY_ROOT}/lib/glfw3.lib" INTERFACE_INCLUDE_DIRECTORIES "${DEPENDENCY_ROOT}/include")
add_library(glad STATIC "${DEPENDENCY_ROOT}/src/glad.c")
target_include_directories(glad PUBLIC "${DEPENDENCY_ROOT}/include")
add_executable(Astana3DTour src/main.cpp)
target_compile_features(Astana3DTour PRIVATE cxx_std_17)
set_target_properties(Astana3DTour PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
target_compile_definitions(Astana3DTour PRIVATE GLFW_INCLUDE_NONE)
target_compile_options(Astana3DTour PRIVATE /W4 /permissive-)
target_link_libraries(Astana3DTour PRIVATE glad glfw_existing opengl32 user32 gdi32 shell32)
add_custom_command(TARGET Astana3DTour POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/shaders" "$<TARGET_FILE_DIR:Astana3DTour>/shaders")
set_property(TARGET Astana3DTour PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "$<TARGET_FILE_DIR:Astana3DTour>")
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT Astana3DTour)
